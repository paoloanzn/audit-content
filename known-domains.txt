# It was generated with the command below; to refresh it, run that with a newer month and replace
# the lines after the last comment.
#   curl -fsSL https://raw.githubusercontent.com/zakird/crux-top-lists/main/data/global/202609.csv.gz | gunzip |
#     awk -F, 'NR > 1 && $2 <= 10000 { sub(/^https?:\/\//, "", $1); sub(/^www\./, "", $1);
